## 🔏 4. 多廠商自適應裁判核心 (`/src/evaluation/judge.py`)

Google Gemini 官方的 OpenAI 相容層（`v1beta/openai/chat/completions`）要求標準的 Bearer 認證。在此命名空間下，Payload 的 `response_format` 必須嚴格對齊 OpenAI 的 `json_object` 規範。

Python

```
"""
Module Name: judge.py
Description: 多廠商自適應非同步裁判核心。
             完美抹平 Gemini 代理層不支援 standard JSON Mode 參數、role: system 與非標準 JSON 響應結構。
Author: Ace (Lead Architect)
"""

import json
import asyncio
import sys
import aiohttp
from pydantic import BaseModel, Field, ValidationError
from tenacity import retry, stop_after_attempt, wait_random_exponential, retry_if_exception_type

class JudgeResponseSchema(BaseModel):
    """強型態裁判回傳防禦閘門"""
    score: int = Field(..., ge=1, le=5, description="嚴格 1 到 5 分的 Rubric 盲測分數")
    reason: str = Field(..., min_length=5, description="具體的判分理由")


class AsyncLLMJudge:
    def __init__(self, provider_type: str, model: str, max_concurrent: int, timeout: float, api_key: str, api_url: str):
        """完全解耦廠商依賴，走原生記憶體定址存取參數，極致最佳化並發執行期效能"""
        self.provider_type = provider_type.lower()
        self.model = model
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.timeout = aiohttp.ClientTimeout(total=timeout)
        self.api_key = api_key
        self.api_url = api_url

    def _prepare_request_meta(self) -> tuple[str, dict]:
        """
        廠商通訊特徵適配器：依據 Vendor 類型自適應建構完全合規的 URL 與 Headers
        完美對齊 Google v1beta/openai/chat/completions 強制要求 Bearer 標頭的資安合規
        """
        headers = {"Content-Type": "application/json"}

        if self.provider_type == "gemini":
            # 🌟 最新對齊：Google 的 OpenAI 相容層已全面對齊標準 Bearer Token 認證
            target_url = self.api_url
            headers["Authorization"] = f"Bearer {self.api_key}"
        else:
            # 標準 OpenAI 官方端點或本地開源 vLLM / Ollama 
            target_url = self.api_url
            headers["Authorization"] = f"Bearer {self.api_key}"

        return target_url, headers

    def _build_rubric_prompt(self, prompt: str, response: str) -> list:
        """
        萬能解耦 Prompt 建構器：
        將 System 裁判規則與 User 數據融合至單一 user 角色中，
        100% 免疫 Google AI Studio 代理層不支援 'role': 'system' 觸發的 400 崩潰漏洞。
        """
        full_instruction = (
            "【系統裁判規則】\n"
            "你是一位極其嚴格的 AI 代碼與邏輯評估裁判。\n"
            "請針對使用者的提示詞與模型輸出的回答進行盲測打分，評分範圍 1 到 5 分：\n"
            "1分：完全離題或產生嚴重幻覺。\n"
            "3分：基本答對，但缺乏深度，或包含些微瑕疵。\n"
            "5分：完美回答，邏輯嚴密，具備生產級參考價值。\n"
            "【注意】你必須嚴格以 JSON 格式回傳，不得包含任何 Markdown 標記（如 ```json）。\n"
            "格式範例：{\"score\": 5, \"reason\": \"原因...\"}\n\n"
            "--------------------------------------------------\n"
            "【待評估盲測數據】\n"
            f"使用者提示詞: {prompt}\n\n"
            f"模型回答內容: {response}"
        )

        return [
            {"role": "user", "content": full_instruction}
        ]

    @retry(
        reraise=True,
        stop=stop_after_attempt(4),
        wait=wait_random_exponential(min=2, max=15),
        retry=retry_if_exception_type((aiohttp.ClientError, asyncio.TimeoutError))
    )
    async def _call_api_with_retry(self, session: aiohttp.ClientSession, payload: dict) -> str:
        # 運行期動態獲取當前廠商的正確標頭與路由
        target_url, headers = self._prepare_request_meta()

        async with session.post(target_url, headers=headers, json=payload, timeout=self.timeout) as response:
            if response.status == 429:
                print("⚠️  [Rate Limit] 觸發 429 限流，看門狗正在調度非同步指數退避重試...")
                raise aiohttp.ClientResponseError(response.request_info, response.history, status=429)

            # 捕捉其餘 HTTP 異常（包含隱蔽的 400/404 封包體內容）
            if response.status != 200:
                err_text = await response.text()
                print(f"❌ [HTTP ERROR DETECTED] 狀態碼: {response.status}, 封包內容: {err_text}", file=sys.stderr)
                response.raise_for_status()

            res_json = await response.json()
            
            # 自適應多型解析核心：平反 Gemini 代理層回傳結構不一致的漏洞
            if "choices" in res_json and len(res_json["choices"]) > 0:
                choice = res_json["choices"][0]
                if "message" in choice and "content" in choice["message"]:
                    return choice["message"]["content"]
            
            if "candidates" in res_json and len(res_json["candidates"]) > 0:
                candidate = res_json["candidates"][0]
                if "content" in candidate and "parts" in candidate["content"]:
                    return candidate["content"]["parts"][0].get("text", "")
            
            raise KeyError(f"無法從多廠商響應體中榨取出有效文本欄位。Raw Response: {res_json}")

    async def evaluate_single_case(self, session: aiohttp.ClientSession, prompt: str, response: str) -> dict:
        """受 Semaphore 保護的單一協程呼叫"""
        async with self.semaphore:

            # 🌟 破口平反：由於 Google 官方新端點已完全對齊標準 OpenAI 規範
            # 這裡無論是 gemini 還是 openai，一律強制使用標準官方指定的 "json_object"
            fmt_type = "json_object"

            payload = {
                "model": self.model,
                "messages": self._build_rubric_prompt(prompt, response),
                "temperature": 0.0,
                "response_format": {"type": fmt_type}
            }

            try:
                raw_content = await self._call_api_with_retry(session, payload)
                
                # 增量防禦：自動剝離 Markdown 包裹標記
                if "```json" in raw_content:
                    raw_content = raw_content.split("```json")[-1].split("```")[0].strip()
                elif "```" in raw_content:
                    raw_content = raw_content.split("```")[-1].split("```")[0].strip()
                    
                data = json.loads(raw_content)

                # Pydantic 守門
                validated = JudgeResponseSchema.model_validate(data)
                return {"score": validated.score, "reason": validated.reason}
            except (ValidationError, json.JSONDecodeError) as e:
                return {"score": 1, "reason": f"裁判模型輸出格式混亂，解析失敗。Raw: {raw_content}. Error: {str(e)}"}
            except Exception as e:
                return {"score": 1, "reason": f"API 呼叫終極失敗: {str(e)}"}
```

## 🌀 5. 序列化評估管線核心 (`/src/evaluation/pipeline.py`)

為了 100% 免疫 Google AI Studio 免費層在微秒級並發下的限流地雷，本模組**廢棄了 `asyncio.gather` 雙發射機制，改採嚴格的序列化 await**。同時，引入「黃金標準答案（Ground Truth Alignment）」架構，將 Baseline 與 Fine-tuned 分別與專家答案進行相似度比對，徹底填補指標漏洞。

Python

```
"""
Module Name: pipeline.py
Description: 基準測試自動化評估管線核心 (黃金標準答案完全體)
Author: Ace (Lead Architect)
"""

import asyncio
import aiohttp
from typing import List, Dict
from src.evaluation.metrics import MetricsCalculator
from src.evaluation.judge import AsyncLLMJudge

class EvaluationPipeline:
    def __init__(self, provider_type: str, model: str, max_concurrent: int, timeout: float, api_key: str, api_url: str):
        self.metrics_calc = MetricsCalculator()
        self.judge = AsyncLLMJudge(
            provider_type=provider_type,
            model=model,
            max_concurrent=max_concurrent,
            timeout=timeout,
            api_key=api_key,
            api_url=api_url
        )

    async def execute(self, dataset: List[Dict[str, str]]) -> List[Dict]:
        print(f"🌀 啟動 Asyncio 事件迴圈，正在調度 {len(dataset)} 筆評估案例...")
        async with aiohttp.ClientSession() as session:
            # 🌟 外層案例同樣走嚴格順序執行，配合 hparams 設為 1，確保多案例間不產生交叉並發
            results = []
            for case in dataset:
                res = await self._evaluate_case_stream(session, case)
                results.append(res)
                # 案例與案例之間額外冷卻 1.0 秒，確保計時器滑動視窗完全降壓
                await asyncio.sleep(1.0)
            return results

    async def _evaluate_case_stream(self, session: aiohttp.ClientSession, case: Dict) -> Dict:
        case_id = case["id"]
        prompt = case["prompt"]
        base_res = case["baseline_response"]
        ft_res = case["finetuned_response"]
        target_res = case.get("target_response", "") # 🌟 引入黃金標準答案

        # 🚀 升級：分別以 target_response 作為 Reference 計算 ROUGE/BLEU 指標
        if target_res.strip():
            base_metrics = self.metrics_calc.compute_string_metrics(reference=target_res, hypothesis=base_res)
            ft_metrics = self.metrics_calc.compute_string_metrics(reference=target_res, hypothesis=ft_res)
        else:
            # Fallback 降級：若無標準答案，則退回原來的雙模型自我對抗模式
            base_metrics = {"rouge1": 0.0, "rouge2": 0.0, "rougeL": 0.0, "bleu": 0.0, "em": 0.0}
            ft_metrics = self.metrics_calc.compute_string_metrics(reference=base_res, hypothesis=ft_res)

        # 🌟 2. 移除 asyncio.gather 破口，強迫雙流序列化前進
        # 先讓 Baseline 裁判請求射擊並安全落地
        judge_base = await self.judge.evaluate_single_case(session, prompt, base_res)
        
        # ⏳ 物理冷卻防護：在兩個請求之間強制掛起 1.5 秒，徹底抹平網關對微秒級併發的敏感度
        await asyncio.sleep(1.5)
        
        # 讓 Fine-tuned 裁判請求射擊並安全落地
        judge_ft = await self.judge.evaluate_single_case(session, prompt, ft_res)

        return {
            "id": case_id,
            "prompt": prompt,
            "base_metrics": base_metrics, # 🌟 拆分獨立指標流
            "ft_metrics": ft_metrics,
            "judge_baseline_score": judge_base["score"],
            "judge_baseline_reason": judge_base["reason"],
            "judge_finetuned_score": judge_ft["score"],
            "judge_finetuned_reason": judge_ft["reason"],
        }
```

## 🏛️ 10 跨模組架構聯動警告：環境變數全域注入之副作用與修復指南

### 🔴 1. 為什麼 `main.py` 和 `run_curation.py` 突然壞了？（成因分析）

在進入模組三之前，專案中的 `main.py` (模組一) 與 `run_curation.py` (模組二) 執行一切正常。然而，在模組三為了引入 Google Gemini 裁判，我們將全專案唯一的中央組態進入點 `config/config.yaml` 升級，加入了環境變數的動態讀取語法：
YAML
```
env:
  PROVIDER_TYPE: "${oc.env:PROVIDER_TYPE}" # 🌟 全域污染源：強迫 Hydra 啟動時必須在系統中抓到此變數
```

這導致了一個嚴重的**全域組態聯動副作用**：

- **Hydra 的全域初始化機制**：不論你單獨執行哪一個指令（`python main.py` 或 `python run_curation.py`），Hydra 在背後**一律都會去加載並編譯同一個全域 `config.yaml`**。

- **記憶體搶佔時間差（死鎖主因）**：當你執行舊程式碼時，由於這兩個檔案的最頂層**沒有載入環境變數**（缺失 `load_dotenv()`），在 Hydra 裝飾器 `@hydra.main` 搶先編譯組態樹的那一微秒，作業系統記憶體裡根本沒有 `PROVIDER_TYPE` 這些變數。

- **結果**：Hydra 找不到全域組態要求的環境變數，直接在程式起跑點丟出 `InterpolationResolutionError` KeyError 熔斷崩潰。
